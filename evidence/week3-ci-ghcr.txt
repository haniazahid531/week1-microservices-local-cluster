===== WORKFLOW FILE =====
name: Week 3 CI and Container Images

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main
  workflow_dispatch:

permissions:
  contents: read
  packages: write

env:
  REGISTRY: ghcr.io
  IMAGE_PREFIX: ghcr.io/${{ github.repository_owner }}/week1-microservices-local-cluster

jobs:
  quality-checks:
    name: Test ${{ matrix.service }}
    runs-on: ubuntu-latest

    strategy:
      fail-fast: false
      matrix:
        service:
          - backend
          - frontend

    steps:
      - name: Checkout repository
        uses: actions/checkout@v6

      - name: Set up Go
        uses: actions/setup-go@v6
        with:
          go-version-file: ${{ matrix.service }}/go.mod
          cache: false

      - name: Check Go formatting
        working-directory: ${{ matrix.service }}
        run: |
          UNFORMATTED="$(gofmt -l .)"

          if [ -n "$UNFORMATTED" ]; then
            echo "The following files require formatting:"
            echo "$UNFORMATTED"
            exit 1
          fi

          echo "Go formatting passed"

      - name: Run tests
        working-directory: ${{ matrix.service }}
        run: go test ./...

      - name: Run Go vet
        working-directory: ${{ matrix.service }}
        run: go vet ./...

      - name: Compile service
        working-directory: ${{ matrix.service }}
        run: go build ./...

  container-images:
    name: Build ${{ matrix.service }} image
    runs-on: ubuntu-latest
    needs: quality-checks

    strategy:
      fail-fast: false
      matrix:
        service:
          - backend
          - frontend

    steps:
      - name: Checkout repository
        uses: actions/checkout@v6

      - name: Set up Docker Buildx
        uses: docker/setup-buildx-action@v4

      - name: Log in to GitHub Container Registry
        if: github.event_name != 'pull_request'
        uses: docker/login-action@v4
        with:
          registry: ${{ env.REGISTRY }}
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}

      - name: Build and push image
        uses: docker/build-push-action@v7
        with:
          context: ./${{ matrix.service }}
          file: ./${{ matrix.service }}/Dockerfile
          push: ${{ github.event_name != 'pull_request' }}
          tags: |
            ${{ env.IMAGE_PREFIX }}-${{ matrix.service }}:${{ github.sha }}
            ${{ env.IMAGE_PREFIX }}-${{ matrix.service }}:latest
          labels: |
            org.opencontainers.image.source=${{ github.server_url }}/${{ github.repository }}
            org.opencontainers.image.revision=${{ github.sha }}
          cache-from: type=gha
          cache-to: type=gha,mode=max
